const tg = window.Telegram.WebApp; tg.ready(); // Legge il parametro 'partite' dall'URL const urlParams = new URLSearchParams(window.location.search); const partiteRaw = urlParams.get('partite'); // Decodifica l'array di partite let listaPartite = []; try { listaPartite = JSON.parse(decodeURIComponent(partiteRaw)); } catch (e) { listaPartite = ["Partita 1", "Partita 2"]; // Fallback di test } // Struttura per salvare le giocate dell'utente: { "Inter vs Milan": "1", "Juventus vs Roma": "X" } let pronosticiScelti = {}; function selezionaEsito(partita, esito) { pronosticiScelti[partita] = esito; console.log("Stato schedina:", pronosticiScelti); } function inviaSchedinaCompleta() { // Controlla che l'utente abbia giocato tutte le partite if (Object.keys(pronosticiScelti).length < listaPartite.length) { alert("Completa i pronostici per tutte le partite prima di inviare!"); return; } // Invia il pacchetto completo a Python const payload = JSON.stringify({ tipo: "schedina_completa", giocate: pronosticiScelti }); tg.sendData(payload); tg.close(); }
